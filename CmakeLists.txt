cmake_minimum_required(VERSION 3.22)
project(ZeroAdventures VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)

foreach(dependency freetype ogg flac vorbis)
    string(TOUPPER "${dependency}" dependency_upper)
    set(local_dependency_dir "${CMAKE_CURRENT_SOURCE_DIR}/external/SFML/build/_deps/${dependency}-src")
    if(EXISTS "${local_dependency_dir}/CMakeLists.txt")
        set("FETCHCONTENT_SOURCE_DIR_${dependency_upper}" "${local_dependency_dir}" CACHE PATH "" FORCE)
    endif()
endforeach()

add_subdirectory(external/SFML)

file(GLOB ZERO_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Zero/*.cpp")

add_executable(Zero ${ZERO_SOURCES})

target_link_libraries(Zero PRIVATE
    sfml-graphics
    sfml-audio
)

add_custom_command(TARGET Zero POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_CURRENT_SOURCE_DIR}/Zero/Data"
            "$<TARGET_FILE_DIR:Zero>/Data"
)
